import { useCallback, useEffect, useRef, useState } from "react";
import * as Speech from "expo-speech";

export default function useSpeech() {
  const [status, setStatus] = useState("idle");
  const [paragraphIndex, setParagraphIndex] = useState(0);

  const [rate, setRate] = useState(1.0);
  const [language, setLanguage] = useState("en-US");

  const [voices, setVoices] = useState([]);
  const [voiceIdentifier, setVoiceIdentifier] = useState(null);
  const [voiceName, setVoiceName] = useState(null);

  const [hasUSVoice, setHasUSVoice] = useState(false);
  const [hasUKVoice, setHasUKVoice] = useState(false);

  const [englishVoices, setEnglishVoices] = useState([]);

  const paragraphsRef = useRef([]);
  const paragraphIndexRef = useRef(0);

  const rateRef = useRef(rate);
  const languageRef = useRef(language);

  const voiceIdentifierRef = useRef(null);
  const voicesRef = useRef([]);

  const commandIdRef = useRef(0);

  // =========================================
  // KEEP CURRENT SETTINGS IN REFS
  // =========================================

  useEffect(() => {
    rateRef.current = rate;
  }, [rate]);

  useEffect(() => {
    languageRef.current = language;
  }, [language]);

  // =========================================
  // LOAD AVAILABLE ANDROID VOICES
  // =========================================

  const loadVoices = useCallback(async () => {
    try {
      const availableVoices = await Speech.getAvailableVoicesAsync();

      const safeVoices = Array.isArray(availableVoices)
        ? availableVoices
        : [];

      const english = safeVoices.filter((voice) => {
        const voiceLanguage = String(voice?.language || "").toLowerCase();

        return (
          voiceLanguage === "en-us" ||
          voiceLanguage === "en-gb"
        );
      });

      const usVoice =
        safeVoices.find(
          (voice) =>
            String(voice?.language || "").toLowerCase() === "en-us",
        ) || null;

      const ukVoice =
        safeVoices.find(
          (voice) =>
            String(voice?.language || "").toLowerCase() === "en-gb",
        ) || null;

      voicesRef.current = safeVoices;

      setVoices(safeVoices);
      setEnglishVoices(english);

      setHasUSVoice(Boolean(usVoice));
      setHasUKVoice(Boolean(ukVoice));

      const currentLanguage = languageRef.current;

      const selectedVoice =
        currentLanguage === "en-GB"
          ? ukVoice
          : usVoice;

      if (selectedVoice) {
        voiceIdentifierRef.current =
          selectedVoice.identifier || null;

        setVoiceIdentifier(
          selectedVoice.identifier || null,
        );

        setVoiceName(
          selectedVoice.name ||
            selectedVoice.identifier ||
            null,
        );
      } else {
        voiceIdentifierRef.current = null;
        setVoiceIdentifier(null);
        setVoiceName(null);
      }

      return safeVoices;
    } catch (error) {
      console.log("Failed to load TTS voices:", error);

      voicesRef.current = [];
      setVoices([]);
      setEnglishVoices([]);
      setHasUSVoice(false);
      setHasUKVoice(false);
      setVoiceIdentifier(null);
      setVoiceName(null);

      return [];
    }
  }, []);

  // =========================================
  // SET PARAGRAPHS
  // =========================================

  const setParagraphs = useCallback((paragraphs) => {
    const safeParagraphs = Array.isArray(paragraphs)
      ? paragraphs
          .filter(
            (paragraph) =>
              typeof paragraph === "string" &&
              paragraph.trim().length > 0,
          )
          .map((paragraph) => paragraph.trim())
      : [];

    paragraphsRef.current = safeParagraphs;

    if (safeParagraphs.length === 0) {
      paragraphIndexRef.current = 0;

      setParagraphIndex(0);
      setStatus("idle");

      return;
    }

    const currentIndex = paragraphIndexRef.current;

    if (currentIndex >= safeParagraphs.length) {
      paragraphIndexRef.current =
        safeParagraphs.length - 1;

      setParagraphIndex(
        safeParagraphs.length - 1,
      );
    }
  }, []);

  // =========================================
  // FIND VOICE
  // =========================================

  const findVoiceForLanguage = useCallback((targetLanguage) => {
    const target = targetLanguage === "en-GB"
      ? "en-gb"
      : "en-us";

    return (
      voicesRef.current.find(
        (voice) =>
          String(voice?.language || "").toLowerCase() ===
          target,
      ) || null
    );
  }, []);

  // =========================================
  // SPEAK CURRENT PARAGRAPH
  // =========================================

  const speakParagraph = useCallback(
    (index) => {
      const paragraphs = paragraphsRef.current;

      if (!paragraphs.length) {
        setStatus("idle");
        return;
      }

      const safeIndex = Math.min(
        Math.max(index, 0),
        paragraphs.length - 1,
      );

      const text = paragraphs[safeIndex]?.trim();

      if (!text) {
        setStatus("idle");
        return;
      }

      const currentCommandId = ++commandIdRef.current;

      paragraphIndexRef.current = safeIndex;
      setParagraphIndex(safeIndex);
      setStatus("speaking");

      const options = {
        language: languageRef.current,
        rate: rateRef.current,
        pitch: 1.0,

        onStart: () => {
          if (currentCommandId !== commandIdRef.current) {
            return;
          }

          setStatus("speaking");
        },

        onDone: () => {
          if (currentCommandId !== commandIdRef.current) {
            return;
          }

          const nextIndex =
            paragraphIndexRef.current + 1;

          if (
            nextIndex <
            paragraphsRef.current.length
          ) {
            speakParagraph(nextIndex);
            return;
          }

          setStatus("finished");
        },

        onStopped: () => {
          if (currentCommandId !== commandIdRef.current) {
            return;
          }

          setStatus("idle");
        },

        onError: (error) => {
          if (currentCommandId !== commandIdRef.current) {
            return;
          }

          console.log("TTS error:", error);
          setStatus("error");
        },
      };

      if (voiceIdentifierRef.current) {
        options.voice =
          voiceIdentifierRef.current;
      }

      Speech.speak(text, options);
    },
    [],
  );

  // =========================================
  // PLAY
  // =========================================

  const play = useCallback(async () => {
    const paragraphs = paragraphsRef.current;

    if (!paragraphs.length) {
      return;
    }

    if (status === "speaking") {
      return;
    }

    try {
      await Speech.stop();
    } catch (error) {
      console.log("TTS stop before play error:", error);
    }

    speakParagraph(paragraphIndexRef.current);
  }, [status, speakParagraph]);

  // =========================================
  // STOP
  // =========================================

  const stop = useCallback(async () => {
    commandIdRef.current += 1;

    try {
      await Speech.stop();
    } catch (error) {
      console.log("TTS stop error:", error);
    }

    setStatus("idle");
  }, []);

  // =========================================
  // PREVIOUS PARAGRAPH
  // =========================================

  const previousParagraph = useCallback(async () => {
    const paragraphs = paragraphsRef.current;

    if (!paragraphs.length) {
      return;
    }

    const previousIndex = Math.max(
      paragraphIndexRef.current - 1,
      0,
    );

    commandIdRef.current += 1;

    try {
      await Speech.stop();
    } catch (error) {
      console.log(
        "TTS previous stop error:",
        error,
      );
    }

    paragraphIndexRef.current = previousIndex;
    setParagraphIndex(previousIndex);

    speakParagraph(previousIndex);
  }, [speakParagraph]);

  // =========================================
  // NEXT PARAGRAPH
  // =========================================

  const nextParagraph = useCallback(async () => {
    const paragraphs = paragraphsRef.current;

    if (!paragraphs.length) {
      return;
    }

    const nextIndex = Math.min(
      paragraphIndexRef.current + 1,
      paragraphs.length - 1,
    );

    commandIdRef.current += 1;

    try {
      await Speech.stop();
    } catch (error) {
      console.log(
        "TTS next stop error:",
        error,
      );
    }

    paragraphIndexRef.current = nextIndex;
    setParagraphIndex(nextIndex);

    speakParagraph(nextIndex);
  }, [speakParagraph]);

  // =========================================
  // CHANGE SPEED
  // =========================================

  const changeRate = useCallback(
    async (newRate) => {
      const safeRate = Math.min(
        Math.max(Number(newRate) || 1.0, 0.5),
        2.0,
      );

      rateRef.current = safeRate;
      setRate(safeRate);

      if (status === "speaking") {
        commandIdRef.current += 1;

        try {
          await Speech.stop();
        } catch (error) {
          console.log(
            "TTS rate change stop error:",
            error,
          );
        }

        speakParagraph(
          paragraphIndexRef.current,
        );
      }
    },
    [status, speakParagraph],
  );

  // =========================================
  // CHANGE LANGUAGE / VOICE
  // =========================================

  const changeLanguage = useCallback(
    async (newLanguage) => {
      const safeLanguage =
        newLanguage === "en-GB"
          ? "en-GB"
          : "en-US";

      languageRef.current = safeLanguage;
      setLanguage(safeLanguage);

      const selectedVoice =
        findVoiceForLanguage(safeLanguage);

      if (selectedVoice) {
        const identifier =
          selectedVoice.identifier || null;

        voiceIdentifierRef.current =
          identifier;

        setVoiceIdentifier(identifier);

        setVoiceName(
          selectedVoice.name ||
            selectedVoice.identifier ||
            null,
        );
      } else {
        voiceIdentifierRef.current = null;

        setVoiceIdentifier(null);
        setVoiceName(null);
      }

      if (status === "speaking") {
        commandIdRef.current += 1;

        try {
          await Speech.stop();
        } catch (error) {
          console.log(
            "TTS language change stop error:",
            error,
          );
        }

        speakParagraph(
          paragraphIndexRef.current,
        );
      }
    },
    [
      findVoiceForLanguage,
      status,
      speakParagraph,
    ],
  );

  // =========================================
  // RESET
  // =========================================

  const reset = useCallback(async () => {
    commandIdRef.current += 1;

    try {
      await Speech.stop();
    } catch (error) {
      console.log("TTS reset error:", error);
    }

    paragraphsRef.current = [];
    paragraphIndexRef.current = 0;

    setParagraphIndex(0);
    setStatus("idle");
  }, []);

  // =========================================
  // LOAD VOICES ON MOUNT
  // =========================================

  useEffect(() => {
    loadVoices();
  }, [loadVoices]);

  // =========================================
  // CLEANUP
  // =========================================

  useEffect(() => {
    return () => {
      commandIdRef.current += 1;

      Speech.stop().catch(() => {});
    };
  }, []);

  // =========================================
  // RETURN
  // =========================================

  return {
    status,
    paragraphIndex,

    rate,
    language,

    voices,
    englishVoices,

    voiceIdentifier,
    voiceName,

    hasUSVoice,
    hasUKVoice,

    setParagraphs,

    play,
    stop,

    previousParagraph,
    nextParagraph,

    changeRate,
    changeLanguage,

    loadVoices,
    reset,
  };
}
