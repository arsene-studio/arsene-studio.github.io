// =================================
// SCREEN
// =================================

export default function ReaderScreen() {
  const {
    id,
    chapter: chapterParam,
    bookmarkProgress: bookmarkProgressParam,
  } = useLocalSearchParams();

  const { colors } = useTheme();

  // =================================
  // LOCAL BOOK
  // =================================

  const localNovel = library[id];

  // =================================
  // REMOTE BOOK
  // =================================

  const [remoteNovel, setRemoteNovel] = useState(null);

  const [remoteLoading, setRemoteLoading] = useState(false);

  const [remoteError, setRemoteError] = useState(null);

  // =================================
  // FINAL NOVEL
  // =================================

  const novel = localNovel || remoteNovel;

  // =================================
  // REFS
  // =================================

  const scrollRef = useRef(null);

  const paragraphPositionsRef = useRef({});

  // =================================
  // READING TIME REFS
  // =================================

  const readingSessionStartRef = useRef(null);

  const readingTimeRef = useRef(0);

  const readingTimeSaveTimerRef = useRef(null);

  const appStateRef = useRef(AppState.currentState);

  const readerFocusedRef = useRef(false);

  // =================================
  // INITIAL CHAPTER
  // =================================

  const initialChapter =
    chapterParam !== undefined ? Number(chapterParam) || 0 : 0;

  // =================================
  // STATE
  // =================================

  const [chapterIndex, setChapterIndex] = useState(initialChapter);

  const [showContents, setShowContents] = useState(false);

  const [showSettings, setShowSettings] = useState(false);

  const [selectedWord, setSelectedWord] = useState(null);

  const [modernWord, setModernWord] = useState(null);

  const [showDictionary, setShowDictionary] = useState(false);

  const [showSearch, setShowSearch] = useState(false);

  const [searchQuery, setSearchQuery] = useState("");

  const [searchResults, setSearchResults] = useState([]);

  const [currentSearchIndex, setCurrentSearchIndex] = useState(0);

  const [highlightArchaic, setHighlightArchaic] = useState(false);

  const [ambientSound, setAmbientSound] = useState("off");

  const [ambientVolume, setAmbientVolume] = useState(0.3);

  // =================================
  // READER CONTROLS
  // =================================

  const [showReaderControls, setShowReaderControls] = useState(true);

  const touchStartY = useRef(0);

  const touchMoved = useRef(false);

  // =================================
  // TTS
  // =================================

  const [showTTSPanel, setShowTTSPanel] = useState(false);

  const {
    status: speechStatus,
    paragraphIndex: speechParagraphIndex,
    rate: speechRate,
    language: speechLanguage,
    hasUSVoice,
    hasUKVoice,
    setParagraphs: setSpeechParagraphs,
    play: playSpeech,
    stop: stopSpeech,
    previousParagraph,
    nextParagraph,
    changeRate,
    changeLanguage,
    reset: resetSpeech,
  } = useSpeech();

  // =================================
  // BOOKMARK
  // =================================

  const [bookmark, setBookmark] = useState(null);

  const [readingMode, setReadingMode] = useState("app");

  const [fontSize, setFontSize] = useState(18);

  const [fontFamily, setFontFamily] = useState("bookerly");

  const [lineSpacing, setLineSpacing] = useState("normal");

  // =================================
  // TEXT ALIGNMENT
  // =================================

  const [textAlignment, setTextAlignment] = useState("left");

  const [scrollProgress, setScrollProgress] = useState(0);

  const scrollProgressRef = useRef(0);

  const [restoreScrollProgress, setRestoreScrollProgress] = useState(null);

  const [contentHeight, setContentHeight] = useState(0);

  const [viewportHeight, setViewportHeight] = useState(0);

  const [hasRestoredPosition, setHasRestoredPosition] = useState(false);

  const [settingsLoaded, setSettingsLoaded] = useState(false);

  const [readingTimeLoaded, setReadingTimeLoaded] = useState(false);

  // =================================
  // KEEP SCROLL PROGRESS REF UPDATED
  // =================================

  useEffect(() => {
    scrollProgressRef.current = scrollProgress;
  }, [scrollProgress]);

  // =================================
  // LOAD REMOTE BOOK
  // =================================

  useEffect(() => {
    let cancelled = false;

    async function loadRemoteBook() {
      if (!id) {
        return;
      }

      if (localNovel) {
        return;
      }

      setRemoteLoading(true);

      setRemoteError(null);

      try {
        const book = await fetchRemoteBookById(id);

        if (cancelled) {
          return;
        }

        if (!book) {
          throw new Error(`Remote book "${id}" was not found.`);
        }

        setRemoteNovel(book);
      } catch (error) {
        if (cancelled) {
          return;
        }

        console.log(`Failed to load remote book "${id}":`, error);

        setRemoteError(error?.message || "Failed to load this book.");
      } finally {
        if (!cancelled) {
          setRemoteLoading(false);
        }
      }
    }

    loadRemoteBook();

    return () => {
      cancelled = true;
    };
  }, [id, localNovel]);

  // =================================
  // READER THEME
  // =================================

  const appReaderTheme = useMemo(
    () => ({
      background: colors.background,
      text: colors.text,
      secondaryText: colors.secondaryText,
      title: colors.title,
      card: colors.card,
      border: colors.border,
      progressTrack: colors.progressTrack,
      progressFill: colors.progressFill,
      iconBackground: colors.iconBackground,
    }),
    [colors],
  );

  const currentTheme =
    readingMode === "app"
      ? appReaderTheme
      : readerThemes[readingMode] || readerThemes.light;

  // =================================
  // FONT
  // =================================

  const currentFont = useMemo(() => {
    return fonts.find((item) => item.id === fontFamily)?.family;
  }, [fontFamily]);

  // =================================
  // SAFE CHAPTER
  // =================================

  const safeChapterIndex = novel?.chapters?.length
    ? Math.min(
        Math.max(chapterIndex, 0),
        Math.max(novel.chapters.length - 1, 0),
      )
    : 0;

  const chapter = novel ? novel.chapters?.[safeChapterIndex] || null : null;

  // =================================
  // CHAPTER STATUS
  // =================================

  const isFirstChapter = safeChapterIndex === 0;

  const isLastChapter = novel
    ? safeChapterIndex === novel.chapters.length - 1
    : true;

  // =================================
  // BOOKMARK STATUS
  // =================================

  const isBookmarked = useMemo(() => {
    if (!bookmark || !chapter) {
      return false;
    }

    if (bookmark.chapterIndex !== safeChapterIndex) {
      return false;
    }

    const savedScrollProgress =
      typeof bookmark.scrollProgress === "number"
        ? bookmark.scrollProgress
        : 0;

    return Math.abs(savedScrollProgress - scrollProgress) <= 0.01;
  }, [bookmark, chapter, safeChapterIndex, scrollProgress]);

  // =================================
  // LOAD READER SETTINGS
  // =================================

  useEffect(() => {
    async function initializeReaderSettings() {
      try {
        const storedSettings = await AsyncStorage.getItem(READER_SETTINGS_KEY);

        if (storedSettings) {
          const settings = JSON.parse(storedSettings);

          if (settings.readingMode) {
            setReadingMode(settings.readingMode);
          }

          if (settings.fontFamily) {
            setFontFamily(settings.fontFamily);
          }

          if (typeof settings.fontSize === "number") {
            setFontSize(settings.fontSize);
          }

          if (settings.lineSpacing) {
            setLineSpacing(settings.lineSpacing);
          }

          if (settings.textAlignment) {
            setTextAlignment(settings.textAlignment);
          }

          if (typeof settings.highlightArchaic === "boolean") {
            setHighlightArchaic(settings.highlightArchaic);
          }

          if (settings.ambientSound) {
            setAmbientSound(settings.ambientSound);
          }

          if (typeof settings.ambientVolume === "number") {
            setAmbientVolume(settings.ambientVolume);
          }
        }
      } catch (error) {
        console.log("Failed to load reader settings:", error);
      } finally {
        setSettingsLoaded(true);
      }
    }

    initializeReaderSettings();
  }, []);

  // =================================
  // SAVE READER SETTINGS
  // =================================

  useEffect(() => {
    if (!settingsLoaded) {
      return;
    }

    async function persistReaderSettings() {
      try {
        const settings = {
          fontSize,
          readingMode,
          fontFamily,
          lineSpacing,
          textAlignment,
          highlightArchaic,
          ambientSound,
          ambientVolume,
        };

        await AsyncStorage.setItem(
          READER_SETTINGS_KEY,
          JSON.stringify(settings),
        );
      } catch (error) {
        console.log("Failed to save reader settings:", error);
      }
    }

    persistReaderSettings();
  }, [
    settingsLoaded,
    fontSize,
    readingMode,
    fontFamily,
    lineSpacing,
    textAlignment,
    highlightArchaic,
    ambientSound,
    ambientVolume,
  ]);

  // =================================
  // LOAD READING TIME
  // =================================

  useEffect(() => {
    let cancelled = false;

    async function loadReadingTime() {
      setReadingTimeLoaded(false);

      readingTimeRef.current = 0;

      try {
        const storedProgress = await AsyncStorage.getItem(READING_PROGRESS_KEY);

        if (cancelled) {
          return;
        }

        if (!storedProgress) {
          return;
        }

        const progressData = JSON.parse(storedProgress);

        const savedBook = progressData[id];

        if (!savedBook) {
          return;
        }

        const savedReadingTime = Number(savedBook.readingTime);

        if (Number.isFinite(savedReadingTime) && savedReadingTime > 0) {
          readingTimeRef.current = Math.floor(savedReadingTime);
        }
      } catch (error) {
        console.log("Failed to load reading time:", error);
      } finally {
        if (!cancelled) {
          setReadingTimeLoaded(true);
        }
      }
    }

    if (id) {
      loadReadingTime();
    }

    return () => {
      cancelled = true;
    };
  }, [id]);

  // =================================
  // QUERY CHAPTER
  // =================================

  useEffect(() => {
    if (chapterParam === undefined) {
      return;
    }

    const newChapterIndex = Number(chapterParam) || 0;

    stopSpeech();

    setShowTTSPanel(false);

    setChapterIndex(newChapterIndex);

    setScrollProgress(0);

    setRestoreScrollProgress(null);

    setHasRestoredPosition(false);

    paragraphPositionsRef.current = {};

    setShowSettings(false);

    setShowContents(false);

    setShowDictionary(false);

    setShowReaderControls(true);

    requestAnimationFrame(() => {
      scrollRef.current?.scrollTo({
        y: 0,
        animated: false,
      });
    });
  }, [chapterParam, stopSpeech]);

  // =================================
  // LOAD BOOKMARK
  // =================================

  useEffect(() => {
    if (!novel) {
      return;
    }

    async function loadBookmarkData() {
      try {
        const storedBookmarks = await AsyncStorage.getItem(BOOKMARKS_KEY);

        if (!storedBookmarks) {
          setBookmark(null);
          return;
        }

        const bookmarks = JSON.parse(storedBookmarks);

        if (!Array.isArray(bookmarks)) {
          setBookmark(null);
          return;
        }

        const savedBookmark = bookmarks.find((item) => item.bookId === id);

        setBookmark(savedBookmark || null);
      } catch (error) {
        console.log("Failed to load bookmark:", error);

        setBookmark(null);
      }
    }

    loadBookmarkData();
  }, [id, novel]);

  // =================================
  // RESTORE BOOKMARK POSITION
  // =================================

  useEffect(() => {
    if (!chapter || !bookmark) {
      return;
    }

    if (chapterParam === undefined) {
      return;
    }

    if (bookmark.chapterIndex !== safeChapterIndex) {
      return;
    }

    const savedScrollProgress =
      typeof bookmark.scrollProgress === "number"
        ? bookmark.scrollProgress
        : 0;

    const safeProgress = Math.min(Math.max(savedScrollProgress, 0), 1);

    setScrollProgress(safeProgress);

    setRestoreScrollProgress(safeProgress);

    setHasRestoredPosition(false);
  }, [chapter, bookmark, chapterParam, safeChapterIndex]);

  // =================================
  // LOAD READING PROGRESS
  // =================================

  useEffect(() => {
    if (!novel) {
      return;
    }

    async function loadProgress() {
      try {
        const storedProgress = await AsyncStorage.getItem(READING_PROGRESS_KEY);

        if (!storedProgress) {
          return;
        }

        const progressData = JSON.parse(storedProgress);

        const savedBook = progressData[id];

        if (!savedBook) {
          return;
        }

        if (chapterParam !== undefined) {
          return;
        }

        const savedChapter = Number(savedBook.chapterIndex) || 0;

        const savedScroll =
          typeof savedBook.scrollProgress === "number"
            ? savedBook.scrollProgress
            : 0;

        setChapterIndex(savedChapter);

        setScrollProgress(savedScroll);

        setRestoreScrollProgress(savedScroll);

        setHasRestoredPosition(false);
      } catch (error) {
        console.log("Failed to load reading progress:", error);
      }
    }

    loadProgress();
  }, [id, chapterParam, novel]);

  // =================================
  // PARAGRAPHS
  // =================================

  const paragraphs = useMemo(() => {
    if (!chapter?.content) {
      return [];
    }

    return String(chapter.content)
      .replace(/\*\*\n/g, "\n")
      .split(/\n\s*\n/)
      .map((paragraph) => paragraph.trim())
      .filter((paragraph) => paragraph.length > 0);
  }, [chapter]);

  // =================================
  // SYNC PARAGRAPHS WITH TTS
  // =================================

  useEffect(() => {
    let active = true;

    const prepareSpeech = async () => {
      await resetSpeech();

      if (active) {
        setSpeechParagraphs(paragraphs);
      }
    };

    prepareSpeech();

    return () => {
      active = false;
    };
  }, [paragraphs, resetSpeech, setSpeechParagraphs]);

  // =================================
  // PARSED PARAGRAPHS
  // =================================

  const parsedParagraphs = useMemo(() => {
    return paragraphs.map((paragraph) => {
      const words = paragraph.split(/(\s+)/);

      const result = [];

      let normalText = "";

      words.forEach((word) => {
        const modern = findModernWord(word);

        if (!modern) {
          normalText += word;

          return;
        }

        if (normalText) {
          result.push({
            type: "normal",
            text: normalText,
          });

          normalText = "";
        }

        result.push({
          type: "archaic",
          word,
          modern,
        });
      });

      if (normalText) {
        result.push({
          type: "normal",
          text: normalText,
        });
      }

      return result;
    });
  }, [paragraphs]);

  // =================================
  // LINE HEIGHT
  // =================================

  const lineHeight =
    lineSpacing === "normal"
      ? Math.round(fontSize * 1.8)
      : Math.round(fontSize * 2);

  // =================================
  // CURRENT READING TIME
  // =================================

  const getCurrentReadingTime = useCallback(() => {
    const baseTime = readingTimeRef.current;

    if (!readingSessionStartRef.current) {
      return baseTime;
    }

    const elapsedSeconds = Math.floor(
      (Date.now() - readingSessionStartRef.current) / 1000,
    );

    return baseTime + Math.max(elapsedSeconds, 0);
  }, []);

  // =================================
  // SAVE READING PROGRESS
  // =================================

  const saveReadingProgress = useCallback(
    async (newChapterIndex, newScrollProgress) => {
      if (!novel || !novel.chapters?.length) {
        return;
      }

      try {
        const storedProgress = await AsyncStorage.getItem(READING_PROGRESS_KEY);

        const progressData = storedProgress ? JSON.parse(storedProgress) : {};

        const existingBook = progressData[id] || {};

        const progress = Math.min(
          Math.max(
            Math.round(
              ((newChapterIndex + newScrollProgress) / novel.chapters.length) *
                100,
            ),
            0,
          ),
          100,
        );

        progressData[id] = {
          ...existingBook,
          bookId: id,
          chapterIndex: newChapterIndex,
          scrollProgress: newScrollProgress,
          progress,
          readingTime: getCurrentReadingTime(),
          updatedAt: Date.now(),
        };

        await AsyncStorage.setItem(
          READING_PROGRESS_KEY,
          JSON.stringify(progressData),
        );
      } catch (error) {
        console.log("Failed to save reading progress:", error);
      }
    },
    [id, novel, getCurrentReadingTime],
  );

  // =================================
  // SAVE READING TIME
  // =================================

  const saveReadingTime = useCallback(
    async (force = false) => {
      if (!id || !novel || !readingTimeLoaded) {
        return;
      }

      if (!readingSessionStartRef.current) {
        return;
      }

      const elapsedSeconds = Math.floor(
        (Date.now() - readingSessionStartRef.current) / 1000,
      );

      if (!force && elapsedSeconds <= 0) {
        return;
      }

      if (elapsedSeconds <= 0) {
        return;
      }

      const newTotalReadingTime = readingTimeRef.current + elapsedSeconds;

      readingTimeRef.current = newTotalReadingTime;

      readingSessionStartRef.current = Date.now();

      try {
        const storedProgress = await AsyncStorage.getItem(READING_PROGRESS_KEY);

        const progressData = storedProgress ? JSON.parse(storedProgress) : {};

        const existingBook = progressData[id] || {};

        progressData[id] = {
          ...existingBook,
          bookId: id,
          readingTime: newTotalReadingTime,
          updatedAt: Date.now(),
        };

        await AsyncStorage.setItem(
          READING_PROGRESS_KEY,
          JSON.stringify(progressData),
        );
      } catch (error) {
        console.log("Failed to save reading time:", error);
      }
    },
    [id, novel, readingTimeLoaded],
  );

  // =================================
  // START READING SESSION
  // =================================

  const startReadingSession = useCallback(() => {
    if (!id || !novel || !readingTimeLoaded) {
      return;
    }

    if (appStateRef.current !== "active") {
      return;
    }

    if (readingSessionStartRef.current) {
      return;
    }

    readingSessionStartRef.current = Date.now();
  }, [id, novel, readingTimeLoaded]);

  // =================================
  // STOP READING SESSION
  // =================================

  const stopReadingSession = useCallback(async () => {
    if (!readingSessionStartRef.current) {
      return;
    }

    await saveReadingTime(true);
  }, [saveReadingTime]);

  // =================================
  // READING TIME AUTO SAVE
  // =================================

  useEffect(() => {
    if (!readingTimeLoaded || !novel) {
      return;
    }

    if (readingTimeSaveTimerRef.current) {
      clearInterval(readingTimeSaveTimerRef.current);

      readingTimeSaveTimerRef.current = null;
    }

    readingTimeSaveTimerRef.current = setInterval(() => {
      if (
        readerFocusedRef.current &&
        appStateRef.current === "active" &&
        readingSessionStartRef.current
      ) {
        saveReadingTime(false);
      }
    }, READING_TIME_SAVE_INTERVAL);

    return () => {
      if (readingTimeSaveTimerRef.current) {
        clearInterval(readingTimeSaveTimerRef.current);

        readingTimeSaveTimerRef.current = null;
      }
    };
  }, [readingTimeLoaded, novel, saveReadingTime]);

  // =================================
  // APP STATE
  // =================================

  useEffect(() => {
    const subscription = AppState.addEventListener(
      "change",
      async (nextAppState) => {
        const previousAppState = appStateRef.current;

        appStateRef.current = nextAppState;

        if (
          previousAppState === "active" &&
          (nextAppState === "inactive" || nextAppState === "background")
        ) {
          stopSpeech();

          await stopReadingSession();

          return;
        }

        if (
          (previousAppState === "inactive" ||
            previousAppState === "background") &&
          nextAppState === "active"
        ) {
          startReadingSession();
        }
      },
    );

    return () => {
      subscription.remove();
    };
  }, [startReadingSession, stopReadingSession, stopSpeech]);

  // =================================
  // READER FOCUS / BLUR
  // =================================

  useFocusEffect(
    useCallback(() => {
      readerFocusedRef.current = true;

      startReadingSession();

      return () => {
        readerFocusedRef.current = false;

        stopSpeech();

        setShowTTSPanel(false);

        stopReadingSession();

        if (novel && chapter) {
          saveReadingProgress(
            safeChapterIndex,
            scrollProgressRef.current,
          );
        }
      };
    }, [
      startReadingSession,
      stopReadingSession,
      stopSpeech,
      novel,
      chapter,
      safeChapterIndex,
      saveReadingProgress,
    ]),
  );

  // =================================
  // RESTORE SCROLL POSITION
  // =================================

  useEffect(() => {
    if (
      restoreScrollProgress === null ||
      contentHeight <= 0 ||
      viewportHeight <= 0 ||
      hasRestoredPosition
    ) {
      return;
    }

    const scrollableHeight = contentHeight - viewportHeight;

    if (scrollableHeight <= 0) {
      setHasRestoredPosition(true);

      return;
    }

    const targetY = scrollableHeight * restoreScrollProgress;

    requestAnimationFrame(() => {
      scrollRef.current?.scrollTo({
        y: targetY,
        animated: false,
      });
    });

    setHasRestoredPosition(true);
  }, [
    restoreScrollProgress,
    contentHeight,
    viewportHeight,
    hasRestoredPosition,
  ]);

  // =================================
  // SHOW READER CONTROLS
  // =================================

  const handleReaderTouchStart = useCallback((event) => {
    touchStartY.current = event.nativeEvent.pageY;

    touchMoved.current = false;
  }, []);

  const handleReaderTouchMove = useCallback((event) => {
    const currentY = event.nativeEvent.pageY;

    if (Math.abs(currentY - touchStartY.current) > 10) {
      touchMoved.current = true;

      setShowReaderControls(false);
    }
  }, []);

  const handleReaderTouchEnd = useCallback(() => {
    if (!touchMoved.current) {
      setShowReaderControls(true);
    }
  }, []);

  // =================================
  // SCROLL
  // =================================

  const handleScroll = useCallback(
    (event) => {
      setShowReaderControls(false);

      if (showSettings) {
        setShowSettings(false);
      }

      if (showDictionary) {
        setShowDictionary(false);
      }

      const { contentOffset, contentSize, layoutMeasurement } =
        event.nativeEvent;

      const scrollableHeight =
        contentSize.height - layoutMeasurement.height;

      if (scrollableHeight <= 0) {
        return;
      }

      const progress = Math.min(
        Math.max(contentOffset.y / scrollableHeight, 0),
        1,
      );

      setScrollProgress(progress);
    },
    [showSettings, showDictionary],
  );

  // =================================
  // SCROLL END
  // =================================

  const handleScrollEnd = useCallback(() => {
    saveReadingProgress(
      safeChapterIndex,
      scrollProgressRef.current,
    );
  }, [safeChapterIndex, saveReadingProgress]);

  // =================================
  // TOGGLE BOOKMARK
  // =================================

  const toggleBookmark = useCallback(async () => {
    if (!chapter || !novel) {
      return;
    }

    try {
      const storedBookmarks = await AsyncStorage.getItem(BOOKMARKS_KEY);

      let bookmarks = storedBookmarks
        ? JSON.parse(storedBookmarks)
        : [];

      if (!Array.isArray(bookmarks)) {
        bookmarks = [];
      }

      const existingBookmarkIndex = bookmarks.findIndex(
        (item) => item.bookId === id,
      );

      // =================================
      // REMOVE CURRENT BOOKMARK
      // =================================

      if (isBookmarked) {
        if (existingBookmarkIndex !== -1) {
          bookmarks.splice(existingBookmarkIndex, 1);
        }

        setBookmark(null);

        await AsyncStorage.setItem(
          BOOKMARKS_KEY,
          JSON.stringify(bookmarks),
        );

        Alert.alert(
          "Bookmark Removed",
          `${chapter.title} has been removed from your bookmark.`,
        );

        return;
      }

      // =================================
      // CREATE / REPLACE BOOKMARK
      // =================================

      const newBookmark = {
        bookId: id,
        chapterId: chapter.id,
        chapterIndex: safeChapterIndex,
        scrollProgress,
        bookTitle: novel.title,
        chapterTitle: chapter.title,
        updatedAt: Date.now(),
      };

      if (existingBookmarkIndex !== -1) {
        bookmarks[existingBookmarkIndex] = newBookmark;
      } else {
        bookmarks.push(newBookmark);
      }

      setBookmark(newBookmark);

      await AsyncStorage.setItem(
        BOOKMARKS_KEY,
        JSON.stringify(bookmarks),
      );

      Alert.alert(
        "Bookmark Saved",
        `${chapter.title} has been saved as your bookmark.`,
      );
    } catch (error) {
      console.log("Failed to save bookmark:", error);
    }
  }, [
    chapter,
    novel,
    id,
    safeChapterIndex,
    scrollProgress,
    isBookmarked,
  ]);

  // =================================
  // TTS CONTROL
  // =================================

  const handleTTSPress = useCallback(() => {
    if (!paragraphs.length) {
      return;
    }

    setShowTTSPanel((previous) => !previous);
  }, [paragraphs.length]);

  // =================================
  // TTS PLAY / PAUSE / RESUME
  // =================================

  const handleTTSPlayPause = useCallback(() => {
    if (!paragraphs.length) {
      return;
    }

    if (speechStatus === "speaking") {
      stopSpeech();

      return;
    }

    playSpeech();
  }, [
    paragraphs.length,
    speechStatus,
    stopSpeech,
    playSpeech,
  ]);

  // =================================
  // TTS PREVIOUS PARAGRAPH
  // =================================

  const handleTTSPreviousParagraph = useCallback(async () => {
    if (!paragraphs.length) {
      return;
    }

    await previousParagraph();
  }, [paragraphs.length, previousParagraph]);

  // =================================
  // TTS NEXT PARAGRAPH
  // =================================

  const handleTTSNextParagraph = useCallback(async () => {
    if (!paragraphs.length) {
      return;
    }

    await nextParagraph();
  }, [paragraphs.length, nextParagraph]);

  // =================================
  // CHAPTER NAVIGATION
  // =================================

  const goToChapter = useCallback(
    (index) => {
      if (!novel) {
        return;
      }

      stopSpeech();

      setShowTTSPanel(false);

      setChapterIndex(index);

      setScrollProgress(0);

      setRestoreScrollProgress(null);

      setHasRestoredPosition(true);

      setShowContents(false);

      setShowSettings(false);

      setShowDictionary(false);

      setShowReaderControls(true);

      paragraphPositionsRef.current = {};

      saveReadingProgress(index, 0);

      requestAnimationFrame(() => {
        scrollRef.current?.scrollTo({
          y: 0,
          animated: true,
        });
      });
    },
    [novel, saveReadingProgress, stopSpeech],
  );

  const goToPreviousChapter = useCallback(() => {
    if (isFirstChapter) {
      return;
    }

    goToChapter(safeChapterIndex - 1);
  }, [
    isFirstChapter,
    safeChapterIndex,
    goToChapter,
  ]);

  const goToNextChapter = useCallback(() => {
    if (isLastChapter) {
      return;
    }

    goToChapter(safeChapterIndex + 1);
  }, [
    isLastChapter,
    safeChapterIndex,
    goToChapter,
  ]);

  // =================================
  // FONT SIZE
  // =================================

  const decreaseFontSize = useCallback(() => {
    setFontSize((previous) => Math.max(14, previous - 2));
  }, []);

  const increaseFontSize = useCallback(() => {
    setFontSize((previous) => Math.min(28, previous + 2));
  }, []);

  // =================================
  // DICTIONARY
  // =================================

  const handleWordPress = useCallback((word) => {
    const modern = findModernWord(word);

    if (!modern) {
      return;
    }

    setSelectedWord(word.replace(/[^a-z']/gi, ""));

    setModernWord(modern);

    setShowDictionary(true);
  }, []);

  // =================================
  // SEARCH
  // =================================

  const handleSearch = useCallback(
    (text) => {
      setSearchQuery(text);

      const query = text.trim().toLowerCase();

      if (!query) {
        setSearchResults([]);

        setCurrentSearchIndex(0);

        return;
      }

      const results = [];

      paragraphs.forEach((paragraph, paragraphIndex) => {
        const lowerParagraph = paragraph.toLowerCase();

        let startIndex = 0;

        while (true) {
          const foundIndex = lowerParagraph.indexOf(
            query,
            startIndex,
          );

          if (foundIndex === -1) {
            break;
          }

          results.push({
            paragraphIndex,
            startIndex: foundIndex,
            endIndex: foundIndex + query.length,
          });

          startIndex = foundIndex + query.length;
        }
      });

      setSearchResults(results);

      setCurrentSearchIndex(0);
    },
    [paragraphs],
  );

  // =================================
  // GO TO SEARCH RESULT
  // =================================

  const goToSearchResult = useCallback(
    (index) => {
      if (searchResults.length === 0) {
        return;
      }

      const safeIndex =
        (index + searchResults.length) %
        searchResults.length;

      const result = searchResults[safeIndex];

      if (!result) {
        return;
      }

      setCurrentSearchIndex(safeIndex);

      const paragraphIndex = result.paragraphIndex;

      const scrollToResult = (attempt = 0) => {
        const y =
          paragraphPositionsRef.current[paragraphIndex];

        if (typeof y !== "number") {
          if (attempt < 10) {
            setTimeout(() => {
              scrollToResult(attempt + 1);
            }, 50);
          }

          return;
        }

        scrollRef.current?.scrollTo({
          y: Math.max(y - 120, 0),
          animated: true,
        });
      };

      requestAnimationFrame(() => {
        scrollToResult();
      });
    },
    [searchResults],
  );

  // =================================
  // PREVIOUS SEARCH RESULT
  // =================================

  const previousSearchResult = useCallback(() => {
    if (searchResults.length === 0) {
      return;
    }

    goToSearchResult(currentSearchIndex - 1);
  }, [
    searchResults.length,
    currentSearchIndex,
    goToSearchResult,
  ]);

  // =================================
  // NEXT SEARCH RESULT
  // =================================

  const nextSearchResult = useCallback(() => {
    if (searchResults.length === 0) {
      return;
    }

    goToSearchResult(currentSearchIndex + 1);
  }, [
    searchResults.length,
    currentSearchIndex,
    goToSearchResult,
  ]);

  // =================================
  // OPEN SEARCH
  // =================================

  const openSearch = useCallback(() => {
    if (showSearch) {
      setShowSearch(false);

      setSearchQuery("");

      setSearchResults([]);

      setCurrentSearchIndex(0);
    } else {
      setShowSearch(true);
    }
  }, [showSearch]);

  // =================================
  // CLOSE SEARCH
  // =================================

  const closeSearch = useCallback(() => {
    setShowSearch(false);

    setSearchQuery("");

    setSearchResults([]);

    setCurrentSearchIndex(0);
  }, []);

  // =================================
  // BOOK LOADING
  // =================================

  if (remoteLoading && !novel) {
    return (
      <SafeAreaView
        style={[
          styles.container,
          {
            backgroundColor: colors.background,
          },
        ]}
        edges={["top", "bottom"]}
      >
        <View style={styles.center}>
          <ActivityIndicator
            size="large"
            color={colors.accent}
          />

          <Text
            style={[
              styles.notFoundTitle,
              {
                color: colors.title,
              },
            ]}
          >
            Loading book...
          </Text>

          <Text
            style={[
              styles.notFoundSubtitle,
              {
                color: colors.secondaryText,
              },
            ]}
          >
            Please wait while the book is being loaded.
          </Text>
        </View>

        <BannerAd />
      </SafeAreaView>
    );
  }

  // =================================
  // BOOK NOT FOUND
  // =================================

  if (!novel) {
    return (
      <SafeAreaView
        style={[
          styles.container,
          {
            backgroundColor: colors.background,
          },
        ]}
        edges={["top", "bottom"]}
      >
        <View style={styles.center}>
          <View
            style={[
              styles.notFoundIcon,
              {
                backgroundColor: colors.chipBackground,
              },
            ]}
          >
            <Ionicons
              name="book-outline"
              size={32}
              color={colors.accent}
            />
          </View>

          <Text
            style={[
              styles.notFoundTitle,
              {
                color: colors.title,
              },
            ]}
          >
            Book not found
          </Text>

          <Text
            style={[
              styles.notFoundSubtitle,
              {
                color: colors.secondaryText,
              },
            ]}
          >
            {remoteError ||
              "This book could not be found in the library."}
          </Text>
        </View>

        <AmbientSound
          sound={ambientSound}
          volume={ambientVolume}
        />

        <BannerAd />
      </SafeAreaView>
    );
  }

  // =================================
  // RENDER
  // =================================

  return (
    <SafeAreaView
      style={[
        styles.container,
        {
          backgroundColor: currentTheme.background,
        },
      ]}
      edges={["top", "bottom"]}
    >
      {/* =================================
          FLOATING TTS
      ================================= */}

      {showReaderControls && (
        <Pressable
          style={({ pressed }) => [
            styles.floatingTTS,
            pressed && {
              opacity: 0.6,
              transform: [
                {
                  scale: 0.95,
                },
              ],
            },
          ]}
          onPress={handleTTSPress}
        >
          <Ionicons
            name="headset-outline"
            size={22}
            color={currentTheme.title}
          />
        </Pressable>
      )}

      {/* =================================
          FLOATING BOOKMARK
      ================================= */}

      {showReaderControls && (
        <Pressable
          style={({ pressed }) => [
            styles.floatingBookmark,
            pressed && {
              opacity: 0.6,
              transform: [
                {
                  scale: 0.95,
                },
              ],
            },
          ]}
          onPress={toggleBookmark}
        >
          <Ionicons
            name={
              isBookmarked
                ? "bookmark"
                : "bookmark-outline"
            }
            size={22}
            color={currentTheme.title}
          />
        </Pressable>
      )}

      {/* =================================
          FLOATING SEARCH
      ================================= */}

      {showReaderControls && (
        <Pressable
          style={({ pressed }) => [
            styles.floatingSearch,
            pressed && {
              opacity: 0.6,
              transform: [
                {
                  scale: 0.95,
                },
              ],
            },
          ]}
          onPress={openSearch}
        >
          <Ionicons
            name="search-outline"
            size={22}
            color={currentTheme.title}
          />
        </Pressable>
      )}

      {/* =================================
          FLOATING SETTINGS
      ================================= */}

      {showReaderControls && (
        <Pressable
          style={({ pressed }) => [
            styles.floatingSettings,
            pressed && {
              opacity: 0.75,
              transform: [
                {
                  scale: 0.95,
                },
              ],
            },
          ]}
          onPress={() =>
            setShowSettings(
              (previous) => !previous,
            )
          }
        >
          <Ionicons
            name="settings-outline"
            size={22}
            color={currentTheme.title}
          />
        </Pressable>
      )}

      {/* =================================
          TTS PANEL
      ================================= */}

      {showTTSPanel && (
        <>
          <Pressable
            style={styles.ttsOverlay}
            onPress={() => setShowTTSPanel(false)}
          />

          <View
            style={[
              styles.ttsPanel,
              {
                backgroundColor: currentTheme.card,
                borderColor: currentTheme.border,
              },
            ]}
          >
            {/* =================================
                TTS HEADER
            ================================= */}

            <View style={styles.ttsHeader}>
              <View
                style={[
                  styles.ttsHeaderIcon,
                  {
                    backgroundColor:
                      currentTheme.iconBackground,
                  },
                ]}
              >
                <Ionicons
                  name="headset-outline"
                  size={18}
                  color={currentTheme.title}
                />
              </View>

              <View style={styles.ttsHeaderText}>
                <Text
                  style={[
                    styles.ttsTitle,
                    {
                      color: currentTheme.title,
                    },
                  ]}
                >
                  Read Aloud
                </Text>

                <Text
                  style={[
                    styles.ttsSubtitle,
                    {
                      color:
                        currentTheme.secondaryText,
                    },
                  ]}
                >
                  {speechStatus === "speaking"
                    ? "Reading..."
                    : speechStatus === "finished"
                      ? "Finished"
                      : "Ready to read"}
                </Text>
              </View>

              <Pressable
                onPress={() =>
                  setShowTTSPanel(false)
                }
                hitSlop={10}
              >
                <Ionicons
                  name="close"
                  size={21}
                  color={currentTheme.secondaryText}
                />
              </Pressable>
            </View>

            {/* =================================
                PLAY / PAUSE
            ================================= */}

            <View style={styles.ttsMainControls}>
              <Pressable
                onPress={handleTTSPreviousParagraph}
                disabled={
                  paragraphs.length === 0 ||
                  speechParagraphIndex <= 0
                }
                style={({ pressed }) => [
                  styles.ttsRoundButton,
                  {
                    backgroundColor:
                      currentTheme.iconBackground,
                  },
                  paragraphs.length === 0 ||
                    speechParagraphIndex <= 0
                    ? styles.ttsDisabledButton
                    : pressed && {
                        opacity: 0.65,
                      },
                ]}
              >
                <Ionicons
                  name="play-skip-back"
                  size={20}
                  color={currentTheme.title}
                />
              </Pressable>

              <Pressable
                onPress={handleTTSPlayPause}
                style={({ pressed }) => [
                  styles.ttsPlayButton,
                  {
                    backgroundColor: NAVY,
                  },
                  pressed && {
                    opacity: 0.8,
                    transform: [
                      {
                        scale: 0.96,
                      },
                    ],
                  },
                ]}
              >
                <Ionicons
                  name={
                    speechStatus === "speaking"
                      ? "pause"
                      : "play"
                  }
                  size={25}
                  color="#FFFFFF"
                />
              </Pressable>

              <Pressable
                onPress={handleTTSNextParagraph}
                disabled={
                  paragraphs.length === 0 ||
                  speechParagraphIndex >=
                    paragraphs.length - 1
                }
                style={({ pressed }) => [
                  styles.ttsRoundButton,
                  {
                    backgroundColor:
                      currentTheme.iconBackground,
                  },
                  paragraphs.length === 0 ||
                    speechParagraphIndex >=
                      paragraphs.length - 1
                    ? styles.ttsDisabledButton
                    : pressed && {
                        opacity: 0.65,
                      },
                ]}
              >
                <Ionicons
                  name="play-skip-forward"
                  size={20}
                  color={currentTheme.title}
                />
              </Pressable>
            </View>

            {/* =================================
                PARAGRAPH INDICATOR
            ================================= */}

            <Text
              style={[
                styles.ttsParagraphIndicator,
                {
                  color: currentTheme.secondaryText,
                },
              ]}
            >
              {paragraphs.length > 0
                ? `Paragraph ${Math.min(
                    speechParagraphIndex + 1,
                    paragraphs.length,
                  )} of ${paragraphs.length}`
                : "No paragraphs available"}
            </Text>

            {/* =================================
                SPEED
            ================================= */}

            <Text
              style={[
                styles.ttsSectionLabel,
                {
                  color: currentTheme.secondaryText,
                },
              ]}
            >
              Speed
            </Text>

            <ScrollView
              horizontal
              showsHorizontalScrollIndicator={false}
              contentContainerStyle={{
                paddingRight: 4,
              }}
            >
              {speechRates.map((item) => (
                <Pressable
                  key={item.value}
                  onPress={() =>
                    changeRate(item.value)
                  }
                  style={({ pressed }) => ({
                    paddingHorizontal: 12,
                    paddingVertical: 8,
                    borderRadius: 18,
                    marginRight: 6,
                    backgroundColor:
                      Math.abs(
                        speechRate - item.value,
                      ) < 0.01
                        ? NAVY
                        : currentTheme.iconBackground,
                    borderWidth: 1,
                    borderColor:
                      Math.abs(
                        speechRate - item.value,
                      ) < 0.01
                        ? NAVY
                        : currentTheme.border,
                    opacity: pressed ? 0.7 : 1,
                  })}
                >
                  <Text
                    style={{
                      color:
                        Math.abs(
                          speechRate - item.value,
                        ) < 0.01
                          ? "#FFFFFF"
                          : currentTheme.title,
                      fontSize: 11,
                      fontWeight: "700",
                    }}
                  >
                    {item.label}
                  </Text>
                </Pressable>
              ))}
            </ScrollView>

            {/* =================================
                VOICE
            ================================= */}

            <Text
              style={[
                styles.ttsSectionLabel,
                {
                  color: currentTheme.secondaryText,
                },
              ]}
            >
              Voice
            </Text>

            <View style={styles.ttsVoiceRow}>
              <Pressable
                disabled={!hasUSVoice}
                onPress={() =>
                  changeLanguage("en-US")
                }
                style={({ pressed }) => [
                  styles.ttsVoiceButton,
                  {
                    backgroundColor:
                      speechLanguage === "en-US"
                        ? NAVY
                        : currentTheme.iconBackground,
                    borderColor:
                      speechLanguage === "en-US"
                        ? NAVY
                        : currentTheme.border,
                  },
                  !hasUSVoice &&
                    styles.ttsDisabledButton,
                  pressed &&
                    hasUSVoice && {
                      opacity: 0.7,
                    },
                ]}
              >
                <Text
                  style={{
                    color:
                      speechLanguage === "en-US"
                        ? "#FFFFFF"
                        : currentTheme.title,
                    fontSize: 12,
                    fontWeight: "700",
                  }}
                >
                  🇺🇸 US English
                </Text>
              </Pressable>

              <Pressable
                disabled={!hasUKVoice}
                onPress={() =>
                  changeLanguage("en-GB")
                }
                style={({ pressed }) => [
                  styles.ttsVoiceButton,
                  {
                    backgroundColor:
                      speechLanguage === "en-GB"
                        ? NAVY
                        : currentTheme.iconBackground,
                    borderColor:
                      speechLanguage === "en-GB"
                        ? NAVY
                        : currentTheme.border,
                  },
                  !hasUKVoice &&
                    styles.ttsDisabledButton,
                  pressed &&
                    hasUKVoice && {
                      opacity: 0.7,
                    },
                ]}
              >
                <Text
                  style={{
                    color:
                      speechLanguage === "en-GB"
                        ? "#FFFFFF"
                        : currentTheme.title,
                    fontSize: 12,
                    fontWeight: "700",
                  }}
                >
                  🇬🇧 UK English
                </Text>
              </Pressable>
            </View>

            {/* =================================
                CHAPTER NAVIGATION
            ================================= */}

            <View style={styles.ttsChapterRow}>
              <Pressable
                onPress={goToPreviousChapter}
                disabled={isFirstChapter}
                style={({ pressed }) => [
                  styles.ttsChapterButton,
                  {
                    backgroundColor:
                      currentTheme.iconBackground,
                    borderColor:
                      currentTheme.border,
                  },
                  isFirstChapter &&
                    styles.ttsDisabledButton,
                  pressed &&
                    !isFirstChapter && {
                      opacity: 0.7,
                    },
                ]}
              >
                <Ionicons
                  name="chevron-back"
                  size={16}
                  color={currentTheme.title}
                />

                <Text
                  style={{
                    color: currentTheme.title,
                    fontSize: 11,
                    fontWeight: "700",
                  }}
                >
                  Previous Chapter
                </Text>
              </Pressable>

              <Pressable
                onPress={goToNextChapter}
                disabled={isLastChapter}
                style={({ pressed }) => [
                  styles.ttsChapterButton,
                  {
                    backgroundColor:
                      currentTheme.iconBackground,
                    borderColor:
                      currentTheme.border,
                  },
                  isLastChapter &&
                    styles.ttsDisabledButton,
                  pressed &&
                    !isLastChapter && {
                      opacity: 0.7,
                    },
                ]}
              >
                <Text
                  style={{
                    color: currentTheme.title,
                    fontSize: 11,
                    fontWeight: "700",
                  }}
                >
                  Next Chapter
                </Text>

                <Ionicons
                  name="chevron-forward"
                  size={16}
                  color={currentTheme.title}
                />
              </Pressable>
            </View>
          </View>
        </>
      )}

      {/* =================================
          SEARCH BAR
      ================================= */}

      {showSearch && (
        <View
          style={[
            styles.searchBar,
            {
              backgroundColor: currentTheme.card,
              borderColor: currentTheme.border,
            },
          ]}
        >
          <Ionicons
            name="search-outline"
            size={20}
            color={currentTheme.secondaryText}
          />

          <TextInput
            style={[
              styles.searchInput,
              {
                color: currentTheme.text,
              },
            ]}
            value={searchQuery}
            onChangeText={handleSearch}
            placeholder="Search in this book..."
            placeholderTextColor={
              currentTheme.secondaryText
            }
            autoFocus
            returnKeyType="search"
          />

          {searchResults.length > 0 && (
            <View style={styles.searchNavigation}>
              <Pressable
                onPress={previousSearchResult}
                hitSlop={8}
                style={({ pressed }) => [
                  styles.searchNavigationButton,
                  pressed && {
                    opacity: 0.5,
                  },
                ]}
              >
                <Ionicons
                  name="chevron-up"
                  size={18}
                  color={currentTheme.title}
                />
              </Pressable>

              <Pressable
                onPress={nextSearchResult}
                hitSlop={8}
                style={({ pressed }) => [
                  styles.searchNavigationButton,
                  pressed && {
                    opacity: 0.5,
                  },
                ]}
              >
                <Ionicons
                  name="chevron-down"
                  size={18}
                  color={currentTheme.title}
                />
              </Pressable>
            </View>
          )}

          {searchQuery.length > 0 && (
            <Text
              style={[
                styles.searchCount,
                {
                  color: currentTheme.secondaryText,
                },
              ]}
            >
              {searchResults.length > 0
                ? `${currentSearchIndex + 1}/${searchResults.length}`
                : "0 results"}
            </Text>
          )}

          <Pressable
            onPress={closeSearch}
            hitSlop={10}
          >
            <Ionicons
              name="close"
              size={21}
              color={currentTheme.secondaryText}
            />
          </Pressable>
        </View>
      )}

      {/* =================================
          PROGRESS BAR
      ================================= */}

      <View
        style={[
          styles.topProgressContainer,
          {
            backgroundColor: currentTheme.card,
            borderBottomColor: currentTheme.border,
          },
        ]}
      >
        <View
          style={[
            styles.topProgressTrack,
            {
              backgroundColor:
                currentTheme.progressTrack,
            },
          ]}
        >
          <View
            style={[
